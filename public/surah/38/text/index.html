بِسۡمِ ٱللَّهِ ٱلرَّحۡمَٰنِ ٱلرَّحِيمِ

  1. ١صٓ وَٱلۡقُرۡءَانِ ذِى ٱلذِّكۡرِ
  2. ٢بَلِ ٱلَّذِينَ كَفَرُواْ فِى عِزَّةٍ وَشِقَاقٍ
  3. ٣كَمۡ أَهۡلَكۡنَا مِن قَبۡلِهِم مِّن قَرۡنٍ فَنَادَواْ وَّلَاتَ حِينَ مَنَاصٍ
  4. ٤وَعَجِبُوٓاْ أَن جَآءَهُم مُّنذِرٌ مِّنۡهُمۡ وَقَالَ ٱلۡكَٰفِرُونَ هَٰذَا سَٰحِرٌ كَذَّابٌ
  5. ٥أَجَعَلَ ٱلۡـَٔالِهَةَ إِلَٰهًا وَٰحِدًا إِنَّ هَٰذَا لَشَىۡءٌ عُجَابٌ
  6. ٦وَٱنطَلَقَ ٱلۡمَلَأُ مِنۡهُمۡ أَنِ ٱمۡشُواْ وَٱصۡبِرُواْ عَلَىٰٓ ءَالِهَتِكُمۡ إِنَّ هَٰذَا لَشَىۡءٌ يُرَادُ
  7. ٧مَا سَمِعۡنَا بِهَٰذَا فِى ٱلۡمِلَّةِ ٱلۡـَٔاخِرَةِ إِنۡ هَٰذَآ إِلَّا ٱخۡتِلَٰقٌ
  8. ٨أَءُنزِلَ عَلَيۡهِ ٱلذِّكۡرُ مِنۢ بَيۡنِنَا بَلۡ هُمۡ فِى شَكٍّ مِّن ذِكۡرِى بَل لَّمَّا يَذُوقُواْ عَذَابِ
  9. ٩أَمۡ عِندَهُمۡ خَزَآئِنُ رَحۡمَةِ رَبِّكَ ٱلۡعَزِيزِ ٱلۡوَهَّابِ
  10. ١٠أَمۡ لَهُم مُّلۡكُ ٱلسَّمَٰوَٰتِ وَٱلۡأَرۡضِ وَمَا بَيۡنَهُمَا فَلۡيَرۡتَقُواْ فِى ٱلۡأَسۡبَٰبِ
  11. ١١جُندٌ مَّا هُنَالِكَ مَهۡزُومٌ مِّنَ ٱلۡأَحۡزَابِ
  12. ١٢كَذَّبَتۡ قَبۡلَهُمۡ قَوۡمُ نُوحٍ وَعَادٌ وَفِرۡعَوۡنُ ذُو ٱلۡأَوۡتَادِ
  13. ١٣وَثَمُودُ وَقَوۡمُ لُوطٍ وَأَصۡحَٰبُ لۡـَٔيۡكَةِ أُوْلَٰٓئِكَ ٱلۡأَحۡزَابُ
  14. ١٤إِن كُلٌّ إِلَّا كَذَّبَ ٱلرُّسُلَ فَحَقَّ عِقَابِ
  15. ١٥وَمَا يَنظُرُ هَٰٓؤُلَآءِ إِلَّا صَيۡحَةً وَٰحِدَةً مَّا لَهَا مِن فَوَاقٍ
  16. ١٦وَقَالُواْ رَبَّنَا عَجِّل لَّنَا قِطَّنَا قَبۡلَ يَوۡمِ ٱلۡحِسَابِ
  17. ١٧ٱصۡبِرۡ عَلَىٰ مَا يَقُولُونَ وَٱذۡكُرۡ عَبۡدَنَا دَاوُۥدَ ذَا ٱلۡأَيۡدِ إِنَّهُۥٓ أَوَّابٌ
  18. ١٨إِنَّا سَخَّرۡنَا ٱلۡجِبَالَ مَعَهُۥ يُسَبِّحۡنَ بِٱلۡعَشِىِّ وَٱلۡإِشۡرَاقِ
  19. ١٩وَٱلطَّيۡرَ مَحۡشُورَةً كُلٌّ لَّهُۥٓ أَوَّابٌ
  20. ٢٠وَشَدَدۡنَا مُلۡكَهُۥ وَءَاتَيۡنَٰهُ ٱلۡحِكۡمَةَ وَفَصۡلَ ٱلۡخِطَابِ
  21. ٢١وَهَلۡ أَتَىٰكَ نَبَؤُاْ ٱلۡخَصۡمِ إِذۡ تَسَوَّرُواْ ٱلۡمِحۡرَابَ
  22. ٢٢إِذۡ دَخَلُواْ عَلَىٰ دَاوُۥدَ فَفَزِعَ مِنۡهُمۡ قَالُواْ لَا تَخَفۡ خَصۡمَانِ بَغَىٰ بَعۡضُنَا عَلَىٰ بَعۡضٍ فَٱحۡكُم بَيۡنَنَا بِٱلۡحَقِّ وَلَا تُشۡطِطۡ وَٱهۡدِنَآ إِلَىٰ سَوَآءِ ٱلصِّرَٰطِ
  23. ٢٣إِنَّ هَٰذَآ أَخِى لَهُۥ تِسۡعٌ وَتِسۡعُونَ نَعۡجَةً وَلِىَ نَعۡجَةٌ وَٰحِدَةٌ فَقَالَ أَكۡفِلۡنِيهَا وَعَزَّنِى فِى ٱلۡخِطَابِ
  24. ٢٤قَالَ لَقَدۡ ظَلَمَكَ بِسُؤَالِ نَعۡجَتِكَ إِلَىٰ نِعَاجِهِۦ وَإِنَّ كَثِيرًا مِّنَ ٱلۡخُلَطَآءِ لَيَبۡغِى بَعۡضُهُمۡ عَلَىٰ بَعۡضٍ إِلَّا ٱلَّذِينَ ءَامَنُواْ وَعَمِلُواْ ٱلصَّٰلِحَٰتِ وَقَلِيلٌ مَّا هُمۡ وَظَنَّ دَاوُۥدُ أَنَّمَا فَتَنَّٰهُ فَٱسۡتَغۡفَرَ رَبَّهُۥ وَخَرَّ رَاكِعًا وَأَنَابَ
  25. ٢٥فَغَفَرۡنَا لَهُۥ ذَٰلِكَ وَإِنَّ لَهُۥ عِندَنَا لَزُلۡفَىٰ وَحُسۡنَ مَـَٔابٍ
  26. ٢٦يَٰدَاوُۥدُ إِنَّا جَعَلۡنَٰكَ خَلِيفَةً فِى ٱلۡأَرۡضِ فَٱحۡكُم بَيۡنَ ٱلنَّاسِ بِٱلۡحَقِّ وَلَا تَتَّبِعِ ٱلۡهَوَىٰ فَيُضِلَّكَ عَن سَبِيلِ ٱللَّهِ إِنَّ ٱلَّذِينَ يَضِلُّونَ عَن سَبِيلِ ٱللَّهِ لَهُمۡ عَذَابٌ شَدِيدٌۢ بِمَا نَسُواْ يَوۡمَ ٱلۡحِسَابِ
  27. ٢٧وَمَا خَلَقۡنَا ٱلسَّمَآءَ وَٱلۡأَرۡضَ وَمَا بَيۡنَهُمَا بَٰطِلًا ذَٰلِكَ ظَنُّ ٱلَّذِينَ كَفَرُواْ فَوَيۡلٌ لِّلَّذِينَ كَفَرُواْ مِنَ ٱلنَّارِ
  28. ٢٨أَمۡ نَجۡعَلُ ٱلَّذِينَ ءَامَنُواْ وَعَمِلُواْ ٱلصَّٰلِحَٰتِ كَٱلۡمُفۡسِدِينَ فِى ٱلۡأَرۡضِ أَمۡ نَجۡعَلُ ٱلۡمُتَّقِينَ كَٱلۡفُجَّارِ
  29. ٢٩كِتَٰبٌ أَنزَلۡنَٰهُ إِلَيۡكَ مُبَٰرَكٌ لِّيَدَّبَّرُوٓاْ ءَايَٰتِهِۦ وَلِيَتَذَكَّرَ أُوْلُواْ ٱلۡأَلۡبَٰبِ
  30. ٣٠وَوَهَبۡنَا لِدَاوُۥدَ سُلَيۡمَٰنَ نِعۡمَ ٱلۡعَبۡدُ إِنَّهُۥٓ أَوَّابٌ
  31. ٣١إِذۡ عُرِضَ عَلَيۡهِ بِٱلۡعَشِىِّ ٱلصَّٰفِنَٰتُ ٱلۡجِيَادُ
  32. ٣٢فَقَالَ إِنِّىٓ أَحۡبَبۡتُ حُبَّ ٱلۡخَيۡرِ عَن ذِكۡرِ رَبِّى حَتَّىٰ تَوَارَتۡ بِٱلۡحِجَابِ
  33. ٣٣رُدُّوهَا عَلَىَّ فَطَفِقَ مَسۡحًۢا بِٱلسُّوقِ وَٱلۡأَعۡنَاقِ
  34. ٣٤وَلَقَدۡ فَتَنَّا سُلَيۡمَٰنَ وَأَلۡقَيۡنَا عَلَىٰ كُرۡسِيِّهِۦ جَسَدًا ثُمَّ أَنَابَ
  35. ٣٥قَالَ رَبِّ ٱغۡفِرۡ لِى وَهَبۡ لِى مُلۡكًا لَّا يَنۢبَغِى لِأَحَدٍ مِّنۢ بَعۡدِىٓ إِنَّكَ أَنتَ ٱلۡوَهَّابُ
  36. ٣٦فَسَخَّرۡنَا لَهُ ٱلرِّيحَ تَجۡرِى بِأَمۡرِهِۦ رُخَآءً حَيۡثُ أَصَابَ
  37. ٣٧وَٱلشَّيَٰطِينَ كُلَّ بَنَّآءٍ وَغَوَّاصٍ
  38. ٣٨وَءَاخَرِينَ مُقَرَّنِينَ فِى ٱلۡأَصۡفَادِ
  39. ٣٩هَٰذَا عَطَآؤُنَا فَٱمۡنُنۡ أَوۡ أَمۡسِكۡ بِغَيۡرِ حِسَابٍ
  40. ٤٠وَإِنَّ لَهُۥ عِندَنَا لَزُلۡفَىٰ وَحُسۡنَ مَـَٔابٍ
  41. ٤١وَٱذۡكُرۡ عَبۡدَنَآ أَيُّوبَ إِذۡ نَادَىٰ رَبَّهُۥٓ أَنِّى مَسَّنِىَ ٱلشَّيۡطَٰنُ بِنُصۡبٍ وَعَذَابٍ
  42. ٤٢ٱرۡكُضۡ بِرِجۡلِكَ هَٰذَا مُغۡتَسَلٌۢ بَارِدٌ وَشَرَابٌ
  43. ٤٣وَوَهَبۡنَا لَهُۥٓ أَهۡلَهُۥ وَمِثۡلَهُم مَّعَهُمۡ رَحۡمَةً مِّنَّا وَذِكۡرَىٰ لِأُوْلِى ٱلۡأَلۡبَٰبِ
  44. ٤٤وَخُذۡ بِيَدِكَ ضِغۡثًا فَٱضۡرِب بِّهِۦ وَلَا تَحۡنَثۡ إِنَّا وَجَدۡنَٰهُ صَابِرًا نِّعۡمَ ٱلۡعَبۡدُ إِنَّهُۥٓ أَوَّابٌ
  45. ٤٥وَٱذۡكُرۡ عِبَٰدَنَآ إِبۡرَٰهِيمَ وَإِسۡحَٰقَ وَيَعۡقُوبَ أُوْلِى ٱلۡأَيۡدِى وَٱلۡأَبۡصَٰرِ
  46. ٤٦إِنَّآ أَخۡلَصۡنَٰهُم بِخَالِصَةٍ ذِكۡرَى ٱلدَّارِ
  47. ٤٧وَإِنَّهُمۡ عِندَنَا لَمِنَ ٱلۡمُصۡطَفَيۡنَ ٱلۡأَخۡيَارِ
  48. ٤٨وَٱذۡكُرۡ إِسۡمَٰعِيلَ وَٱلۡيَسَعَ وَذَا ٱلۡكِفۡلِ وَكُلٌّ مِّنَ ٱلۡأَخۡيَارِ
  49. ٤٩هَٰذَا ذِكۡرٌ وَإِنَّ لِلۡمُتَّقِينَ لَحُسۡنَ مَـَٔابٍ
  50. ٥٠جَنَّٰتِ عَدۡنٍ مُّفَتَّحَةً لَّهُمُ ٱلۡأَبۡوَٰبُ
  51. ٥١مُتَّكِـِٔينَ فِيهَا يَدۡعُونَ فِيهَا بِفَٰكِهَةٍ كَثِيرَةٍ وَشَرَابٍ
  52. ٥٢وَعِندَهُمۡ قَٰصِرَٰتُ ٱلطَّرۡفِ أَتۡرَابٌ
  53. ٥٣هَٰذَا مَا تُوعَدُونَ لِيَوۡمِ ٱلۡحِسَابِ
  54. ٥٤إِنَّ هَٰذَا لَرِزۡقُنَا مَا لَهُۥ مِن نَّفَادٍ
  55. ٥٥هَٰذَا وَإِنَّ لِلطَّٰغِينَ لَشَرَّ مَـَٔابٍ
  56. ٥٦جَهَنَّمَ يَصۡلَوۡنَهَا فَبِئۡسَ ٱلۡمِهَادُ
  57. ٥٧هَٰذَا فَلۡيَذُوقُوهُ حَمِيمٌ وَغَسَّاقٌ
  58. ٥٨وَءَاخَرُ مِن شَكۡلِهِۦٓ أَزۡوَٰجٌ
  59. ٥٩هَٰذَا فَوۡجٌ مُّقۡتَحِمٌ مَّعَكُمۡ لَا مَرۡحَبًۢا بِهِمۡ إِنَّهُمۡ صَالُواْ ٱلنَّارِ
  60. ٦٠قَالُواْ بَلۡ أَنتُمۡ لَا مَرۡحَبًۢا بِكُمۡ أَنتُمۡ قَدَّمۡتُمُوهُ لَنَا فَبِئۡسَ ٱلۡقَرَارُ
  61. ٦١قَالُواْ رَبَّنَا مَن قَدَّمَ لَنَا هَٰذَا فَزِدۡهُ عَذَابًا ضِعۡفًا فِى ٱلنَّارِ
  62. ٦٢وَقَالُواْ مَا لَنَا لَا نَرَىٰ رِجَالًا كُنَّا نَعُدُّهُم مِّنَ ٱلۡأَشۡرَارِ
  63. ٦٣أَتَّخَذۡنَٰهُمۡ سِخۡرِيًّا أَمۡ زَاغَتۡ عَنۡهُمُ ٱلۡأَبۡصَٰرُ
  64. ٦٤إِنَّ ذَٰلِكَ لَحَقٌّ تَخَاصُمُ أَهۡلِ ٱلنَّارِ
  65. ٦٥قُلۡ إِنَّمَآ أَنَا۠ مُنذِرٌ وَمَا مِنۡ إِلَٰهٍ إِلَّا ٱللَّهُ ٱلۡوَٰحِدُ ٱلۡقَهَّارُ
  66. ٦٦رَبُّ ٱلسَّمَٰوَٰتِ وَٱلۡأَرۡضِ وَمَا بَيۡنَهُمَا ٱلۡعَزِيزُ ٱلۡغَفَّٰرُ
  67. ٦٧قُلۡ هُوَ نَبَؤٌاْ عَظِيمٌ
  68. ٦٨أَنتُمۡ عَنۡهُ مُعۡرِضُونَ
  69. ٦٩مَا كَانَ لِىَ مِنۡ عِلۡمٍۭ بِٱلۡمَلَإِ ٱلۡأَعۡلَىٰٓ إِذۡ يَخۡتَصِمُونَ
  70. ٧٠إِن يُوحَىٰٓ إِلَىَّ إِلَّآ أَنَّمَآ أَنَا۠ نَذِيرٌ مُّبِينٌ
  71. ٧١إِذۡ قَالَ رَبُّكَ لِلۡمَلَٰٓئِكَةِ إِنِّى خَٰلِقٌۢ بَشَرًا مِّن طِينٍ
  72. ٧٢فَإِذَا سَوَّيۡتُهُۥ وَنَفَخۡتُ فِيهِ مِن رُّوحِى فَقَعُواْ لَهُۥ سَٰجِدِينَ
  73. ٧٣فَسَجَدَ ٱلۡمَلَٰٓئِكَةُ كُلُّهُمۡ أَجۡمَعُونَ
  74. ٧٤إِلَّآ إِبۡلِيسَ ٱسۡتَكۡبَرَ وَكَانَ مِنَ ٱلۡكَٰفِرِينَ
  75. ٧٥قَالَ يَٰٓإِبۡلِيسُ مَا مَنَعَكَ أَن تَسۡجُدَ لِمَا خَلَقۡتُ بِيَدَىَّ أَسۡتَكۡبَرۡتَ أَمۡ كُنتَ مِنَ ٱلۡعَالِينَ
  76. ٧٦قَالَ أَنَا۠ خَيۡرٌ مِّنۡهُ خَلَقۡتَنِى مِن نَّارٍ وَخَلَقۡتَهُۥ مِن طِينٍ
  77. ٧٧قَالَ فَٱخۡرُجۡ مِنۡهَا فَإِنَّكَ رَجِيمٌ
  78. ٧٨وَإِنَّ عَلَيۡكَ لَعۡنَتِىٓ إِلَىٰ يَوۡمِ ٱلدِّينِ
  79. ٧٩قَالَ رَبِّ فَأَنظِرۡنِىٓ إِلَىٰ يَوۡمِ يُبۡعَثُونَ
  80. ٨٠قَالَ فَإِنَّكَ مِنَ ٱلۡمُنظَرِينَ
  81. ٨١إِلَىٰ يَوۡمِ ٱلۡوَقۡتِ ٱلۡمَعۡلُومِ
  82. ٨٢قَالَ فَبِعِزَّتِكَ لَأُغۡوِيَنَّهُمۡ أَجۡمَعِينَ
