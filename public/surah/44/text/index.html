بِسۡمِ ٱللَّهِ ٱلرَّحۡمَٰنِ ٱلرَّحِيمِ

  1. ١حمٓ
  2. ٢وَٱلۡكِتَٰبِ ٱلۡمُبِينِ
  3. ٣إِنَّآ أَنزَلۡنَٰهُ فِى لَيۡلَةٍ مُّبَٰرَكَةٍ إِنَّا كُنَّا مُنذِرِينَ
  4. ٤فِيهَا يُفۡرَقُ كُلُّ أَمۡرٍ حَكِيمٍ
  5. ٥أَمۡرًا مِّنۡ عِندِنَآ إِنَّا كُنَّا مُرۡسِلِينَ
  6. ٦رَحۡمَةً مِّن رَّبِّكَ إِنَّهُۥ هُوَ ٱلسَّمِيعُ ٱلۡعَلِيمُ
  7. ٧رَبِّ ٱلسَّمَٰوَٰتِ وَٱلۡأَرۡضِ وَمَا بَيۡنَهُمَآ إِن كُنتُم مُّوقِنِينَ
  8. ٨لَآ إِلَٰهَ إِلَّا هُوَ يُحۡىِۦ وَيُمِيتُ رَبُّكُمۡ وَرَبُّ ءَابَآئِكُمُ ٱلۡأَوَّلِينَ
  9. ٩بَلۡ هُمۡ فِى شَكٍّ يَلۡعَبُونَ
  10. ١٠فَٱرۡتَقِبۡ يَوۡمَ تَأۡتِى ٱلسَّمَآءُ بِدُخَانٍ مُّبِينٍ
  11. ١١يَغۡشَى ٱلنَّاسَ هَٰذَا عَذَابٌ أَلِيمٌ
  12. ١٢رَّبَّنَا ٱكۡشِفۡ عَنَّا ٱلۡعَذَابَ إِنَّا مُؤۡمِنُونَ
  13. ١٣أَنَّىٰ لَهُمُ ٱلذِّكۡرَىٰ وَقَدۡ جَآءَهُمۡ رَسُولٌ مُّبِينٌ
  14. ١٤ثُمَّ تَوَلَّوۡاْ عَنۡهُ وَقَالُواْ مُعَلَّمٌ مَّجۡنُونٌ
  15. ١٥إِنَّا كَاشِفُواْ ٱلۡعَذَابِ قَلِيلًا إِنَّكُمۡ عَآئِدُونَ
  16. ١٦يَوۡمَ نَبۡطِشُ ٱلۡبَطۡشَةَ ٱلۡكُبۡرَىٰٓ إِنَّا مُنتَقِمُونَ
  17. ١٧وَلَقَدۡ فَتَنَّا قَبۡلَهُمۡ قَوۡمَ فِرۡعَوۡنَ وَجَآءَهُمۡ رَسُولٌ كَرِيمٌ
  18. ١٨أَنۡ أَدُّوٓاْ إِلَىَّ عِبَادَ ٱللَّهِ إِنِّى لَكُمۡ رَسُولٌ أَمِينٌ
  19. ١٩وَأَن لَّا تَعۡلُواْ عَلَى ٱللَّهِ إِنِّىٓ ءَاتِيكُم بِسُلۡطَٰنٍ مُّبِينٍ
  20. ٢٠وَإِنِّى عُذۡتُ بِرَبِّى وَرَبِّكُمۡ أَن تَرۡجُمُونِ
  21. ٢١وَإِن لَّمۡ تُؤۡمِنُواْ لِى فَٱعۡتَزِلُونِ
  22. ٢٢فَدَعَا رَبَّهُۥٓ أَنَّ هَٰٓؤُلَآءِ قَوۡمٌ مُّجۡرِمُونَ
  23. ٢٣فَأَسۡرِ بِعِبَادِى لَيۡلًا إِنَّكُم مُّتَّبَعُونَ
  24. ٢٤وَٱتۡرُكِ ٱلۡبَحۡرَ رَهۡوًا إِنَّهُمۡ جُندٌ مُّغۡرَقُونَ
  25. ٢٥كَمۡ تَرَكُواْ مِن جَنَّٰتٍ وَعُيُونٍ
  26. ٢٦وَزُرُوعٍ وَمَقَامٍ كَرِيمٍ
  27. ٢٧وَنَعۡمَةٍ كَانُواْ فِيهَا فَٰكِهِينَ
  28. ٢٨كَذَٰلِكَ وَأَوۡرَثۡنَٰهَا قَوۡمًا ءَاخَرِينَ
  29. ٢٩فَمَا بَكَتۡ عَلَيۡهِمُ ٱلسَّمَآءُ وَٱلۡأَرۡضُ وَمَا كَانُواْ مُنظَرِينَ
  30. ٣٠وَلَقَدۡ نَجَّيۡنَا بَنِىٓ إِسۡرَٰٓءِيلَ مِنَ ٱلۡعَذَابِ ٱلۡمُهِينِ
  31. ٣١مِن فِرۡعَوۡنَ إِنَّهُۥ كَانَ عَالِيًا مِّنَ ٱلۡمُسۡرِفِينَ
  32. ٣٢وَلَقَدِ ٱخۡتَرۡنَٰهُمۡ عَلَىٰ عِلۡمٍ عَلَى ٱلۡعَٰلَمِينَ
  33. ٣٣وَءَاتَيۡنَٰهُم مِّنَ ٱلۡـَٔايَٰتِ مَا فِيهِ بَلَٰٓؤٌاْ مُّبِينٌ
  34. ٣٤إِنَّ هَٰٓؤُلَآءِ لَيَقُولُونَ
  35. ٣٥إِنۡ هِىَ إِلَّا مَوۡتَتُنَا ٱلۡأُولَىٰ وَمَا نَحۡنُ بِمُنشَرِينَ
  36. ٣٦فَأۡتُواْ بِـَٔابَآئِنَآ إِن كُنتُمۡ صَٰدِقِينَ
  37. ٣٧أَهُمۡ خَيۡرٌ أَمۡ قَوۡمُ تُبَّعٍ وَٱلَّذِينَ مِن قَبۡلِهِمۡ أَهۡلَكۡنَٰهُمۡ إِنَّهُمۡ كَانُواْ مُجۡرِمِينَ
  38. ٣٨وَمَا خَلَقۡنَا ٱلسَّمَٰوَٰتِ وَٱلۡأَرۡضَ وَمَا بَيۡنَهُمَا لَٰعِبِينَ
  39. ٣٩مَا خَلَقۡنَٰهُمَآ إِلَّا بِٱلۡحَقِّ وَلَٰكِنَّ أَكۡثَرَهُمۡ لَا يَعۡلَمُونَ
  40. ٤٠إِنَّ يَوۡمَ ٱلۡفَصۡلِ مِيقَٰتُهُمۡ أَجۡمَعِينَ
  41. ٤١يَوۡمَ لَا يُغۡنِى مَوۡلًى عَن مَّوۡلًى شَيۡـًٔا وَلَا هُمۡ يُنصَرُونَ
  42. ٤٢إِلَّا مَن رَّحِمَ ٱللَّهُ إِنَّهُۥ هُوَ ٱلۡعَزِيزُ ٱلرَّحِيمُ
  43. ٤٣إِنَّ شَجَرَتَ ٱلزَّقُّومِ
  44. ٤٤طَعَامُ ٱلۡأَثِيمِ
  45. ٤٥كَٱلۡمُهۡلِ يَغۡلِى فِى ٱلۡبُطُونِ
  46. ٤٦كَغَلۡىِ ٱلۡحَمِيمِ
  47. ٤٧خُذُوهُ فَٱعۡتِلُوهُ إِلَىٰ سَوَآءِ ٱلۡجَحِيمِ
  48. ٤٨ثُمَّ صُبُّواْ فَوۡقَ رَأۡسِهِۦ مِنۡ عَذَابِ ٱلۡحَمِيمِ
  49. ٤٩ذُقۡ إِنَّكَ أَنتَ ٱلۡعَزِيزُ ٱلۡكَرِيمُ
  50. ٥٠إِنَّ هَٰذَا مَا كُنتُم بِهِۦ تَمۡتَرُونَ
  51. ٥١إِنَّ ٱلۡمُتَّقِينَ فِى مَقَامٍ أَمِينٍ
  52. ٥٢فِى جَنَّٰتٍ وَعُيُونٍ
  53. ٥٣يَلۡبَسُونَ مِن سُندُسٍ وَإِسۡتَبۡرَقٍ مُّتَقَٰبِلِينَ
  54. ٥٤كَذَٰلِكَ وَزَوَّجۡنَٰهُم بِحُورٍ عِينٍ
  55. ٥٥يَدۡعُونَ فِيهَا بِكُلِّ فَٰكِهَةٍ ءَامِنِينَ
  56. ٥٦لَا يَذُوقُونَ فِيهَا ٱلۡمَوۡتَ إِلَّا ٱلۡمَوۡتَةَ ٱلۡأُولَىٰ وَوَقَىٰهُمۡ عَذَابَ ٱلۡجَحِيمِ
  57. ٥٧فَضۡلًا مِّن رَّبِّكَ ذَٰلِكَ هُوَ ٱلۡفَوۡزُ ٱلۡعَظِيمُ
  58. ٥٨فَإِنَّمَا يَسَّرۡنَٰهُ بِلِسَانِكَ لَعَلَّهُمۡ يَتَذَكَّرُونَ
  59. ٥٩فَٱرۡتَقِبۡ إِنَّهُم مُّرۡتَقِبُونَ
