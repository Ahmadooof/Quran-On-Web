بِسۡمِ ٱللَّهِ ٱلرَّحۡمَٰنِ ٱلرَّحِيمِ

  1. ١وَٱلصَّٰٓفَّٰتِ صَفًّا

  5. ٥رَّبُّ ٱلسَّمَٰوَٰتِ وَٱلۡأَرۡضِ وَمَا بَيۡنَهُمَا وَرَبُّ ٱلۡمَشَٰرِقِ

  11. ١١فَٱسۡتَفۡتِهِمۡ أَهُمۡ أَشَدُّ خَلۡقًا أَم مَّنۡ خَلَقۡنَآ إِنَّا خَلَقۡنَٰهُم مِّن طِينٍ لَّازِبٍۭ

  15. ١٥وَقَالُوٓاْ إِنۡ هَٰذَآ إِلَّا سِحۡرٌ مُّبِينٌ
  16. ١٦أَءِذَا مِتۡنَا وَكُنَّا تُرَابًا وَعِظَٰمًا أَءِنَّا لَمَبۡعُوثُونَ
  17. ١٧أَوَءَابَآؤُنَا ٱلۡأَوَّلُونَ

  22. ٢٢ٱحۡشُرُواْ ٱلَّذِينَ ظَلَمُواْ وَأَزۡوَٰجَهُمۡ وَمَا كَانُواْ يَعۡبُدُونَ
  23. ٢٣مِن دُونِ ٱللَّهِ فَٱهۡدُوهُمۡ إِلَىٰ صِرَٰطِ ٱلۡجَحِيمِ

  28. ٢٨قَالُوٓاْ إِنَّكُمۡ كُنتُمۡ تَأۡتُونَنَا عَنِ ٱلۡيَمِينِ

  30. ٣٠وَمَا كَانَ لَنَا عَلَيۡكُم مِّن سُلۡطَٰنٍۭ بَلۡ كُنتُمۡ قَوۡمًا طَٰغِينَ
  31. ٣١فَحَقَّ عَلَيۡنَا قَوۡلُ رَبِّنَآ إِنَّا لَذَآئِقُونَ
  32. ٣٢فَأَغۡوَيۡنَٰكُمۡ إِنَّا كُنَّا غَٰوِينَ
  33. ٣٣فَإِنَّهُمۡ يَوۡمَئِذٍ فِى ٱلۡعَذَابِ مُشۡتَرِكُونَ

  126. ١٢٦ٱللَّهَ رَبَّكُمۡ وَرَبَّ ءَابَآئِكُمُ ٱلۡأَوَّلِينَ
  127. ١٢٧فَكَذَّبُوهُ فَإِنَّهُمۡ لَمُحۡضَرُونَ
  128. ١٢٨إِلَّا عِبَادَ ٱللَّهِ ٱلۡمُخۡلَصِينَ
  129. ١٢٩وَتَرَكۡنَا عَلَيۡهِ فِى ٱلۡـَٔاخِرِينَ
  130. ١٣٠سَلَٰمٌ عَلَىٰٓ إِلۡ يَاسِينَ
  131. ١٣١إِنَّا كَذَٰلِكَ نَجۡزِى ٱلۡمُحۡسِنِينَ
  132. ١٣٢إِنَّهُۥ مِنۡ عِبَادِنَا ٱلۡمُؤۡمِنِينَ
  133. ١٣٣وَإِنَّ لُوطًا لَّمِنَ ٱلۡمُرۡسَلِينَ
  134. ١٣٤إِذۡ نَجَّيۡنَٰهُ وَأَهۡلَهُۥٓ أَجۡمَعِينَ
  135. ١٣٥إِلَّا عَجُوزًا فِى ٱلۡغَٰبِرِينَ
  136. ١٣٦ثُمَّ دَمَّرۡنَا ٱلۡـَٔاخَرِينَ
  137. ١٣٧وَإِنَّكُمۡ لَتَمُرُّونَ عَلَيۡهِم مُّصۡبِحِينَ
  138. ١٣٨وَبِٱلَّيۡلِ أَفَلَا تَعۡقِلُونَ
  139. ١٣٩وَإِنَّ يُونُسَ لَمِنَ ٱلۡمُرۡسَلِينَ
  140. ١٤٠إِذۡ أَبَقَ إِلَى ٱلۡفُلۡكِ ٱلۡمَشۡحُونِ
  141. ١٤١فَسَاهَمَ فَكَانَ مِنَ ٱلۡمُدۡحَضِينَ
  142. ١٤٢فَٱلۡتَقَمَهُ ٱلۡحُوتُ وَهُوَ مُلِيمٌ
  143. ١٤٣فَلَوۡلَآ أَنَّهُۥ كَانَ مِنَ ٱلۡمُسَبِّحِينَ
  144. ١٤٤لَلَبِثَ فِى بَطۡنِهِۦٓ إِلَىٰ يَوۡمِ يُبۡعَثُونَ
  145. ١٤٥فَنَبَذۡنَٰهُ بِٱلۡعَرَآءِ وَهُوَ سَقِيمٌ
  146. ١٤٦وَأَنۢبَتۡنَا عَلَيۡهِ شَجَرَةً مِّن يَقۡطِينٍ
  147. ١٤٧وَأَرۡسَلۡنَٰهُ إِلَىٰ مِاْئَةِ أَلۡفٍ أَوۡ يَزِيدُونَ
  148. ١٤٨فَـَٔامَنُواْ فَمَتَّعۡنَٰهُمۡ إِلَىٰ حِينٍ
  149. ١٤٩فَٱسۡتَفۡتِهِمۡ أَلِرَبِّكَ ٱلۡبَنَاتُ وَلَهُمُ ٱلۡبَنُونَ
  150. ١٥٠أَمۡ خَلَقۡنَا ٱلۡمَلَٰٓئِكَةَ إِنَٰثًا وَهُمۡ شَٰهِدُونَ
  151. ١٥١أَلَآ إِنَّهُم مِّنۡ إِفۡكِهِمۡ لَيَقُولُونَ
  152. ١٥٢وَلَدَ ٱللَّهُ وَإِنَّهُمۡ لَكَٰذِبُونَ
  153. ١٥٣أَصۡطَفَى ٱلۡبَنَاتِ عَلَى ٱلۡبَنِينَ
  154. ١٥٤مَا لَكُمۡ كَيۡفَ تَحۡكُمُونَ
  155. ١٥٥أَفَلَا تَذَكَّرُونَ
  156. ١٥٦أَمۡ لَكُمۡ سُلۡطَٰنٌ مُّبِينٌ
  157. ١٥٧فَأۡتُواْ بِكِتَٰبِكُمۡ إِن كُنتُمۡ صَٰدِقِينَ
  158. ١٥٨وَجَعَلُواْ بَيۡنَهُۥ وَبَيۡنَ ٱلۡجِنَّةِ نَسَبًا وَلَقَدۡ عَلِمَتِ ٱلۡجِنَّةُ إِنَّهُمۡ لَمُحۡضَرُونَ
  159. ١٥٩سُبۡحَٰنَ ٱللَّهِ عَمَّا يَصِفُونَ
  160. ١٦٠إِلَّا عِبَادَ ٱللَّهِ ٱلۡمُخۡلَصِينَ
  161. ١٦١فَإِنَّكُمۡ وَمَا تَعۡبُدُونَ
  162. ١٦٢مَآ أَنتُمۡ عَلَيۡهِ بِفَٰتِنِينَ
  163. ١٦٣إِلَّا مَنۡ هُوَ صَالِ ٱلۡجَحِيمِ
  164. ١٦٤وَمَا مِنَّآ إِلَّا لَهُۥ مَقَامٌ مَّعۡلُومٌ
  165. ١٦٥وَإِنَّا لَنَحۡنُ ٱلصَّآفُّونَ
  166. ١٦٦وَإِنَّا لَنَحۡنُ ٱلۡمُسَبِّحُونَ
  167. ١٦٧وَإِن كَانُواْ لَيَقُولُونَ
  168. ١٦٨لَوۡ أَنَّ عِندَنَا ذِكۡرًا مِّنَ ٱلۡأَوَّلِينَ
  169. ١٦٩لَكُنَّا عِبَادَ ٱللَّهِ ٱلۡمُخۡلَصِينَ
  170. ١٧٠فَكَفَرُواْ بِهِۦ فَسَوۡفَ يَعۡلَمُونَ
  171. ١٧١وَلَقَدۡ سَبَقَتۡ كَلِمَتُنَا لِعِبَادِنَا ٱلۡمُرۡسَلِينَ
  172. ١٧٢إِنَّهُمۡ لَهُمُ ٱلۡمَنصُورُونَ
  173. ١٧٣وَإِنَّ جُندَنَا لَهُمُ ٱلۡغَٰلِبُونَ
  174. ١٧٤فَتَوَلَّ عَنۡهُمۡ حَتَّىٰ حِينٍ
  175. ١٧٥وَأَبۡصِرۡهُمۡ فَسَوۡفَ يُبۡصِرُونَ
  176. ١٧٦أَفَبِعَذَابِنَا يَسۡتَعۡجِلُونَ
  177. ١٧٧فَإِذَا نَزَلَ بِسَاحَتِهِمۡ فَسَآءَ صَبَاحُ ٱلۡمُنذَرِينَ
  178. ١٧٨وَتَوَلَّ عَنۡهُمۡ حَتَّىٰ حِينٍ
  179. ١٧٩وَأَبۡصِرۡ فَسَوۡفَ يُبۡصِرُونَ
  180. ١٨٠سُبۡحَٰنَ رَبِّكَ رَبِّ ٱلۡعِزَّةِ عَمَّا يَصِفُونَ
  181. ١٨١وَسَلَٰمٌ عَلَى ٱلۡمُرۡسَلِينَ
  182. ١٨٢وَٱلۡحَمۡدُ لِلَّهِ رَبِّ ٱلۡعَٰلَمِينَ
